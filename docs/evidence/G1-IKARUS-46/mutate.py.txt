def run_suites() -> tuple[int, str]:
    proc = subprocess.run([str(PY), "-m", "pytest", "-q", "-p", "no:cacheprovider", *SUITES, "--no-header", "-rf"],
                          cwd=ROOT, capture_output=True, text=True, encoding="utf-8", errors="replace")
    return proc.returncode, proc.stdout + proc.stderr


def main() -> int:
    rows = []
    for label, rel, old, new in MUTATIONS:
        path = ROOT / rel
        # BYTES, not text: ``write_text`` on Windows rewrites LF as CRLF and
        # would move the byte-pinned modules (Odysseus 2026-09-10, defect 6).
        original_bytes = path.read_bytes()
        original = original_bytes.decode("utf-8")
        if original.count(old) != 1:
            rows.append((label, "NOT APPLIED (anchor not unique)", ""))
            continue
        path.write_bytes(original.replace(old, new, 1).encode("utf-8"))
        try:
            code, out = run_suites()
        finally:
            path.write_bytes(original_bytes)
        failed = sorted({line.split(" - ")[0].replace("FAILED ", "").strip()
                         for line in out.splitlines() if line.startswith("FAILED ") or line.startswith("ERROR ")})
        summary = [line for line in out.splitlines() if " passed" in line or " failed" in line or " error" in line]
        rows.append((label, "CAUGHT" if code != 0 else "SURVIVED", "; ".join(failed[:6]) + (" | " + summary[-1] if summary else "")))
    code, out = run_suites()
    rows.append(("restored tree", "GREEN" if code == 0 else "RED", [l for l in out.splitlines() if " passed" in l][-1:] and out.splitlines()[-1]))
    for label, verdict, detail in rows:
        print(f"{verdict:9} | {label} | {detail}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
